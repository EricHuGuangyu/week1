SELECT  c.Id AS [Customer Id], c.FirstName, c.LastName, { fn CONCAT(c.FirstName + ' ', c.LastName) } AS [Full Name], s.DateSold, 
                   p.Name, p.Price, dbo.GetCustomerAmount(c.Id) AS [Total Purchases], dbo.CalculateAge(c.DateOfBirth) AS Age, 
                   { fn CONCAT(st.Name + ' ', st.Location) } AS Store
FROM      dbo.Customer AS c LEFT OUTER JOIN
                   dbo.Sale AS s ON c.Id = s.CustomerId LEFT OUTER JOIN
                   dbo.Product AS p ON s.ProductId = p.Id LEFT OUTER JOIN
                   dbo.Store AS st ON st.Id = s.StoreId

ALTER FUNCTION [dbo].[CalculateAge] (@BirthDate DATE)
RETURNS INT
AS
BEGIN
    RETURN DATEDIFF(YEAR, @BirthDate, GETDATE()) 
           - CASE 
               WHEN DATEADD(YEAR, DATEDIFF(YEAR, @BirthDate, GETDATE()), @BirthDate) > GETDATE() 
               THEN 1 
               ELSE 0 
             END;
END;
